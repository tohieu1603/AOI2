            string sendJson = JsonConvert.SerializeObject(data);
            string sendUrl = DataGroup.GetGroupData("MesConfig", "SaveTestDataURL").m_dbOffset.S;

            bool success = HttpHelper.Post(sendUrl, sendJson, out strRes);

            RecordMgr.SaveTxtLog(RecordMode.Mes_SaveTestData, sendUrl, sendJson, strRes);

            strMsg = reciveData?.message ?? strRes;

        }


        public static bool SaveMesData(MesUploadDataModel data, out string strMsg)
        {
            if(!Parameter.GetBit("MesEnable"))
            {
                strMsg = "Disable Mes";
                return true;
            }

            var payload = new
            {
                reportTime = data.reportTime,
                domainId = Parameter.GetString("domainId"),

                machineNo = Parameter.GetString("eqpId"),
                woId = data.woId,
                model = Parameter.GetString("model"),
                testStartTime = data.testStartTime,
                testEndTime = data.testEndTime,
                executionDuration = data.executionDuration,
                inputQty = data.inputQty,
                totalPassQty = data.totalPassQty,
                totalFailQty = data.totalFailQty,
                itemList = data.itemList.Select(x => new
                {
                    defectCode = x.errorCode,
                    defectName = x.errorDesc,
                    qty = x.failQty
                })
            };

            var sendJson = JsonConvert.SerializeObject(payload);
            var url = DataGroup.GetGroupData("MesConfig", "SaveTestDataURL").m_dbOffset.S;
            string strRes = "";
            bool success = HttpHelper.Post(url, sendJson, out strRes);

            var reciveData = JsonConvert.DeserializeObject<MesResponseModel>(strRes);
            bool bRet = success && (reciveData?.success ?? false);
            strMsg = reciveData?.message ?? strRes;

            RecordMgr.SaveTxtLog(RecordMode.Mes_SaveTestData, url, sendJson, strRes);
            return bRet;
        }
