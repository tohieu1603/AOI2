🏗️ 2. Ba cấp độ giải pháp kỹ thuật

  • Cấp 1: Sửa nhanh (Giảm 70-80% giật) — Bỏ Refresh(), thêm SuspendLayout, bật DoubleBuffered, inline counter.
  • Cấp 2: Virtual Mode (Giảm 95%+ giật) — DataGridView không tạo Row object, chỉ hỏi dữ liệu khi cần vẽ.
  • Cấp 3: Virtual Mode + Throttle Timer (Real-time 100%) — Gom các event thành 1 nhịp vẽ (30-50 FPS), triệt tiêu hoàn toàn giật lag.

Code hoàn chỉnh Ring Buffer O(1):

public class ProductDataRingBuffer
{
    private readonly string[][] _buffer;
    private readonly Color[] _foreColors;
    private readonly int _capacity;
    private int _head;
    private int _count;

    public ProductDataRingBuffer(int capacity)
    {
        _capacity = capacity;
        _buffer = new string[capacity][];
        _foreColors = new Color[capacity];
        _head = 0;
        _count = 0;
    }

    public int Count => _count;

    public void Push(string[] values, Color foreColor)
    {
        _buffer[_head] = values;
        _foreColors[_head] = foreColor;
        _head = (_head + 1) % _capacity;
        if (_count < _capacity) _count++;
    }

    public string GetCellValue(int displayRow, int col)
    {
        int idx = ((_head - 1 - displayRow) % _capacity + _capacity) % _capacity;
        var row = _buffer[idx];
        if (row == null || col >= row.Length) return "";
        return row[col] ?? "";
    }

    public Color GetForeColor(int displayRow)
    {
        int idx = ((_head - 1 - displayRow) % _capacity + _capacity) % _capacity;
        return _foreColors[idx];
    }

    public void Clear()
    {
        _count = 0;
        _head = 0;
        Array.Clear(_buffer, 0, _capacity);
        Array.Clear(_foreColors, 0, _capacity);
    }
}
