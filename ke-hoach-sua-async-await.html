// 1. Khoi tao HttpClient duy nhat dung chung
private static readonly HttpClient _httpClient = new HttpClient
{
    Timeout = TimeSpan.FromSeconds(10)
};

// 2. Ham Async duy nhat - Khong ton tai ham wrapper dong bo
public static async Task<(bool success, string message)> SaveTestDataAsync(MesUploadDataModel data)
{
    if (!Parameter.GetBit("MesEnable"))
        return (true, "Disable Mes");

    data.domainId = Parameter.GetString("domainId");
    data.machineNo = Parameter.GetString("machineNo");
    data.model = Parameter.GetString("model");

    string sendJson = JsonConvert.SerializeObject(data);
    string sendUrl = DataGroup.GetGroupData("MesConfig", "SaveTestDataURL").m_dbOffset.S;
    string strRes = "";
    bool isHttpOk = false;

    try
    {
        var content = new StringContent(sendJson, Encoding.UTF8, "application/json");
        // ASYNC: Khong giu chan luong CPU
        HttpResponseMessage response = await _httpClient.PostAsync(sendUrl, content).ConfigureAwait(false);
        strRes = await response.Content.ReadAsStringAsync().ConfigureAwait(false);
        isHttpOk = response.IsSuccessStatusCode;
    }
    catch (Exception ex)
    {
        strRes = ex.Message;
        isHttpOk = false;
    }

    var reciveData = JsonConvert.DeserializeObject<MesResponseModel<object>>(strRes);
    bool bRet = isHttpOk && (reciveData?.success ?? false);
    string strMsg = reciveData?.message ?? strRes;

    RecordMgr.SaveTxtLog(RecordMode.Mes_SaveTestData, sendUrl, sendJson, strRes);
    return (bRet, strMsg);
}

private static readonly SemaphoreSlim _sfcLock = new SemaphoreSlim(1, 1);

public static async Task<(bool success, string send, string receive)> SendAndReceiveAsync(SFCMode mode, string cardId)
{
    string strSend = "";
    string strReceive = "";
    if (cardId != null) cardId = cardId.Replace("+", "%2B");

    // Dinh dang chuoi gui theo mode (giu nguyen)
    // ...

    await _sfcLock.WaitAsync().ConfigureAwait(false);
    try
    {
        int failCount = 0;
        while (true)
        {
            failCount++;
            try
            {
                Connect();
                SendMessage(strSend);
                // DUNG TASK.DELAY: Giai phong luong CPU trong 100ms
                await Task.Delay(100).ConfigureAwait(false);
                link.ReadLine(out strReceive, "}@\n");
                break;
            }
            catch
            {
                Disconnect();
                await Task.Delay(200).ConfigureAwait(false);
            }

            if (failCount >= 3)
            {
                Disconnect();
                return (false, strSend, strReceive);
            }
        }
    }
    finally
    {
        _sfcLock.Release();
    }

    return (true, strSend.Trim('\n'), (strReceive ?? "").Trim('\n'));
}

private async Task UploadHiveDataSafelyAsync()
{
    try
    {
        HiveTool.MachineDataParameters parameters = new HiveTool.MachineDataParameters();
        switch (StationEx.Process)
        {
            case AutoProcess.正常生产:
                parameters.Mode = Parameter.GetBit("Enable_PDCA") 
                    ? ProcessingMode.NormalRunWithPdcaSfc 
                    : ProcessingMode.NormalRunNotWithPdcaSfc;
                break;
            case AutoProcess.空跑:
                parameters.Mode = ProcessingMode.DryRun;
                break;
            case AutoProcess.CPK:
                parameters.Mode = ProcessingMode.CPK;
                break;
            default: break;
        }

        // Chay bat dong bo va bat ngoai le neu co
        await Task.Run(() => HiveMgr.OnSendProductData(parameters)).ConfigureAwait(false);
    }
    catch (Exception ex)
    {
        Logger.Process("HIVE", "Loi upload Hive: " + ex.Message);
    }
}

private static readonly SemaphoreSlim _logSemaphore = new SemaphoreSlim(1, 1);

public static async Task SaveTxtLogAsync(RecordMode mode, string url, string request, string response)
{
    try
    {
        string dir = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "Log", "MesLog");
        if (!Directory.Exists(dir))
        {
            Directory.CreateDirectory(dir);
        }

        string strPath = Path.Combine(dir, $"MesLog_{DateTime.Now:yyyyMMdd}.txt");
        string strLog = $"{DateTime.Now:yyyy-MM-dd HH:mm:ss:fff} -----> {mode} --> {url}\\r\\n"
                      + $"REQUEST  ---> {request}\\r\\n"
                      + $"RESPONSE ---> {response}\\r\\n\\r\\n";

        // Khoa bat dong bo tranh xung dot ghi file
        await _logSemaphore.WaitAsync().ConfigureAwait(false);
        try
        {
            byte[] encodedText = Encoding.UTF8.GetBytes(strLog);
            using (var sourceStream = new FileStream(strPath, 
                FileMode.Append, FileAccess.Write, FileShare.Read, 
                bufferSize: 4096, useAsync: true))
            {
                await sourceStream.WriteAsync(encodedText, 0, encodedText.Length).ConfigureAwait(false);
            }
        }
        finally
        {
            _logSemaphore.Release();
        }
    }
    catch (Exception ex)
    {
        Console.WriteLine("Loi ghi log: " + ex.Message);
    }
}
Ket qua dat duoc:
Ghi file thuc hien truc tiep qua I/O Completion Ports (IOCP) cua he dieu hanh, khong gay nghe luong CPU luc he thong dang ghi file dung luong lon.

CAC VI TRI TUYET DOI KHONG SU DUNG ASYNC / AWAIT TRONG DU AN NAY

  • StationUnloadBinEx.ProduceRun va cac lop ke thua StationBase: Vong lap chuyen buoc san xuat override string ProduceRun(string strStep) la ham dong bo duoc thiet ke boi DLL framework AF.Station.dll. Khong duoc sua chu ky tram thanh async vi se lam vo co che Template Method.
  • Cac ham cho phan cung (WaitAxis, WaitCylIn, WaitCylOut, WaitRegBit): Cac ham nay can do doc tin hieu I/O card lien tuc tren luong Dedicated rieng cua may. Su dung async tai day se lam sai lech do tre thoi gian thuc cua cac sensor va motor.
  • Nguyen tac tranh Deadlock: Khi goi ham async trong WinForms (Form UI), khong bao gio duoc dung .Result hoac .GetAwaiter().GetResult() tren luong chinh UI. Hay tuan thu nguyen tac dung tu khoa await.
